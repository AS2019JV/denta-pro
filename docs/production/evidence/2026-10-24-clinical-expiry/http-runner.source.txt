async function probe(f){

  const r=await http(route,{method,headers:f.headers,body}),raw=await r.text();let value;try{value=JSON.parse(raw);}catch{}

  if(!passed)return{checks,allPassed:false};
 }
 return{checks,allPassed:true};
}
